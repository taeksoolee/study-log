의사코드
function maxSum(arr, k) {
  // 첫 윈도우 합 계산
  sum = arr[0..k-1].sum()
  maxSum = sum
  maxStart = 0
  for i = 1; i <= n-k; i++ {
    // 윈도우 이동: 앞 빼고 뒤 추가
    sum += arr[i+k-1]
    sum -= arr[i-1]
    if sum > maxSum {
      maxSum = sum
      maxStart = i
    }
  }
  return maxSum, maxStart
}
복잡도
시간: O(n)
공간: O(1)
브루트 포스 O(n·k)에 비해
훨씬 효율적입니다.

function lengthOfLongestSubstring(s) {
  charSet = new Set()
  left = 0, maxLen = 0
  for right = 0; right < n; right++ {
    // 중복 문자면 왼쪽 포인터 이동
    while s[right] in charSet {
      charSet.delete(s[left])
      left++
    }
    // 새 문자 추가
    charSet.add(s[right])
    maxLen = max(maxLen, right - left + 1)
  }
  return maxLen
}
포인터 범례
L — 왼쪽 포인터 (left)
R — 오른쪽 포인터 (right)
파란색 — 현재 윈도우
노란색 — 중복 발견
복잡도
시간: O(n)
공간: O(min(m,n))
m: 문자 집합 크기
